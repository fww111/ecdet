cmake_minimum_required(VERSION 3.16)
project(spacecraft_sample C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
file(GLOB SAMPLE_SOURCES CONFIGURE_DEPENDS src/*.c)
add_executable(spacecraft ${SAMPLE_SOURCES})
target_include_directories(spacecraft PRIVATE include)
if(NOT MSVC)
  target_compile_options(spacecraft PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(spacecraft PRIVATE m)
endif()
